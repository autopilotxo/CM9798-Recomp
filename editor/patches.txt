# CM2ED.EXE (the Database Editor, rt/editor.c): single-instruction patches, as patches.txt is for the game.
#   <address> replace|after <C statement>

# Sharp text: as each character is about to be drawn, like the game's 01a70a - CM2ED has the same
# text routine (0x21254), with its variables elsewhere (textlayer.c: text_use_editor)
0212d2 after   hook_text_glyph();

# One click, one action: the editor takes a button that is down as a click, and now opens its next
# screen long before the button is let go - so the same click also chose something on that screen
# (a record's field, straight after the record). Its menus (0x1AE50, called again and again until
# something is chosen: the choice is in AL at the end, -1 for none) no longer see a button still
# down from a press that has already chosen something; its text boxes (0x1B5E4) wait for it to be
# let go before they start.
01ae63 after   hook_ed_menu_begin();
01b1a5 after   hook_ed_menu_end();
01b5f1 after   hook_ed_fresh_click();

# Main menu: "Back to main menu" instead of "Exit program" (and its question and title), and a line
# saying which database is being edited above it (texts in rt/editor.c)
01059c replace ebx = ed_text_back();
0105f9 replace ebx = ed_text_back_question();
010603 replace eax = ed_text_back_title();
0105a8 after   hook_ed_caption();

# Load Database: choose Players, Managers or Teams (PLAYERS.DB1, MGDATA.DB1, TMDATA.DB1) from a
# list instead of typing the file's name. The text box (0x1B5E4, "ret 4") is replaced: the name
# comes back in EAX as it did, and the box's Escape flag (0x10A89AF) says whether it was cancelled.
010f67 replace eax = ed_pick_file(); esp += 4;

# Save Database: no file name to type - it saves the file that was loaded (its name is at 0x1081328,
# which the name box started with), after a Yes/No "Save all changes?" (the box that asked
# "File already exists. Overwrite ?", now always asked: 0x23E00 is "does the file exist?").
# With nothing loaded, the editor still says so itself.
0117f4 replace eax = ed_save_name(); esp += 4;
01185d replace eax = 1;
01186b replace ebx = ed_text_save_question();
